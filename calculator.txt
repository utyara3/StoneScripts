  "7", "8", "9", "/",

  "0", "<", "=", "-"
]

var all_ops = ["+", "-", "*", "/"]

func calculate_string(equation)
  var nums = []
  var ops = []
  var cur = ""
  
  for i = 0..string.Size(equation)-1
    var c = string.Sub(equation, i, 1)
    ?all_ops.Contains(c)
      nums.Add(int.Parse(cur))
      ops.Add(c)
      cur = ""
    :
      cur = cur + c
  nums.Add(int.Parse(cur))  
  
  var new_nums = [nums[0]] 
  var new_ops = []
  
  for i = 0..ops.Count()-1
    ?ops[i] = "*"
      var last_idx = new_nums.Count() - 1
      new_nums[last_idx] = new_nums[last_idx] * nums[i+1]
    :?ops[i] = "/"
      var last_idx = new_nums.Count() - 1
      new_nums[last_idx] = new_nums[last_idx] / nums[i+1]
    :
      new_ops.Add(ops[i])
      new_nums.Add(nums[i+1])
  
  var result = new_nums[0]
  
  ?new_nums.Count() > 1
    for i = 0..new_ops.Count()-1
      ?new_ops[i] = "+"
        result = result + new_nums[i+1]
      :
        result = result - new_nums[i+1]
  
  return result + ""

func handle_press(btn)
  var text = btn.text

  ?string.Size(GetTyped()) = 0 // проверяем чтобы первый был не знак
    for op : all_ops
      ?text = op
        return

  ?text = "="
    var solved_eq = calculate_string(GetTyped())

    pressed.Clear()
    pressed.Add(solved_eq)
    output.text = solved_eq
  :?text = "C" // для того чтобы убрать все, но я урбал эту кнопку из за дизайна
    pressed.Clear()
    output.text = GetTyped()
  :?text = "<"
    pressed.RemoveAt(pressed.Count()-1)
    output.text = GetTyped()
  :
    ButtonPressed(btn)

func ButtonPressed(btn)
  pressed.Add(btn.text)
  output.text = GetTyped()

func GetTyped()
  return string.Join("", pressed)
