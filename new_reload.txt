func update_smooth_track()

    smooth_track［0］ = pos.x
    smooth_track［1］ = 0
  :
    smooth_track［1］ = smooth_track［1］ + 1

var components
func check_weapon_name(weap, short)
  components = string.Split(short, true)
  for c : components
    ?string.Sub(c, 0, 1) = ＂-＂
      ?weap = string.Sub(c, 1)
        return false
    :
      ?weap ! c
        return false

  return true

func reload_listen()
  ?ranged.Contains(item.right)
  ^ & item.right.state ! 4
    ranged.RemoveAt(ranged.IndexOf(item.right))
  ?melee.Contains(item.right)
  ^ & item.right.state ! 4
    melee.RemoveAt(melee.IndexOf(item.right))

  ?ranged.Contains(item.left)
  ^ & item.left.state ! 4
    ranged.RemoveAt(ranged.IndexOf(item.left))
  ?melee.Contains(item.left)
  ^ & item.left.state ! 4
    melee.RemoveAt(melee.IndexOf(item.left))

  ?item.right.state = 3
    ?reload_valid(item.right)
      ?reload_type(item.right) = ＂ranged＂
        ?!ranged.Contains(item.right)
          ranged.Add(item.right)
      :
        ?!melee.Contains(item.right)
          melee.Add(item.right)

  ?item.left.state = 3
    ?reload_valid(item.left)
      ?reload_type(item.left) = ＂ranged＂
        ?!ranged.Contains(item.left)
          ranged.Add(item.left)
      :
        ?!melee.Contains(item.left)
          melee.Add(item.left)

func can_reload_melee()
  ?melee.Count() > 0
    return true
  return false

func can_reload_ranged()
  ?ranged.Count() > 0
    return true
  return false

func can_reload()
  return can_reload_melee() | can_reload_ranged()

func need_reload(weapon)
  for m : melee
    ?check_weapon_name(m, weapon)
      return true
  for r : ranged
    ?check_weapon_name(r, weapon)
      return true
  return false

var ret_cond
var weap_need_reload
func next_weapon(weapons, num)
  ret_list.Clear()
  for weapon : weapons
    weap_need_reload = need_reload(weapon)
    ret_cond = num - (weapons.Count() -
    ^weapons.IndexOf(weapon)) >= 0

    ?ret_list.Count() = num
      break

    for one_hand : one_handed_weaps
      ?check_weapon_name(weapon, one_hand)
        ?!weap_need_reload | ret_cond
          ret_list.Clear()
          ret_list.Add(weapon)

          return ret_list

    ?!weap_need_reload | ret_cond
      ret_list.Add(weapon)

  return ret_list

func reload_all()
  ?can_reload_melee()
    reload_melee()
  :?can_reload_ranged()
    reload_ranged()


func moondial(s1, s2)
  ?!need_reload(s1)
    equipL moon
    equipR @s1@
  :?!need_reload(s2)
    equipL moon
    equipR @s2@
  :
    equipR @s1@
    equipL @s2@


func burst(items)
  var next_weaps = next_weapon(items, 2)
  ?next_weaps.Count() > 1
    equipR @next_weaps[0]@
    equipL @next_weaps[1]@
  :
    equip @next_weaps[0]@
