func format_weapon(weapon)
  ret_weap = ＂＂

  for rune : weapons_rune
    ?weapon = rune
      ret_weap += string.Capitalize(rune) + ＂ ＂
  
  for type : weapons_type
    ?weapon = type
      ret_weap += string.Capitalize(type) + ＂ ＂
  
  ?weapon = ＂max＂
    ret_weap += ＂*10* ＂
  :
    ret_weap += ＂*＂+string.Split(weapon, ＂*＂)［1］
    ^ + ＂* ＂

  ret_weap += ＂+＂ + string.Split(
  ^weapon, ＂+＂)［1］    
  
  return ret_weap


var components
func check_weapon_name(weap, short)
  components = string.Split(short, true)
  for c : components
    ?string.Sub(c, 0, 1) = ＂-＂
      ?weap = string.Sub(c, 1)
        return false
    :
      ?weap ! c
        return false

  return true

func reload_listen()

  ?ranged.Contains(item.right) 
  ^ & item.right.state ! 4
    ranged.RemoveAt(ranged.IndexOf(item.right))
  ?melee.Contains(item.right)
  ^ & item.right.state ! 4
    melee.RemoveAt(melee.IndexOf(item.right))
  
  ?ranged.Contains(item.left)
  ^ & item.left.state ! 4
    ranged.RemoveAt(ranged.IndexOf(item.left))
  ?melee.Contains(item.left)
  ^ & item.left.state ! 4
    melee.RemoveAt(melee.IndexOf(item.left))

  ?item.right.state = 3
    ?reload_valid(item.right)
      ?reload_type(item.right) = ＂ranged＂
        ?!ranged.Contains(item.right)
          ranged.Add(item.right)
      :
        ?!melee.Contains(item.right)
          melee.Add(item.right)

  ?item.left.state = 3
    ?reload_valid(item.left)
      ?reload_type(item.left) = ＂ranged＂
        ?!ranged.Contains(item.left)
          ranged.Add(item.left)
      :
        ?!melee.Contains(item.left)
          melee.Add(item.left)     

func can_reload_melee()
  ?melee.Count() > 0
    return true
  return false

func can_reload_ranged()
  ?ranged.Count() > 0
    return true
  return false

func can_reload()
  ?can_reload_melee() | can_reload_ranged()
    return true
  return false

func need_reload(weapon)
  for m : melee
    ?check_weapon_name(m, weapon)
      return true
  for r : ranged
    ?check_weapon_name(r, weapon)
      return true
  return false

  ^(item.GetCooldown(item) <= 0

func binSearchEnemy(min, max, foe_cnt)
  ?min = max
    return min
  var mid = (min + max)/2
  ?foe.GetCount(mid) < foe_cnt
    return binSearchEnemy(mid + 1, max, foe_cnt)
  :
    return binSearchEnemy(min, mid, foe_cnt)

  return binSearchEnemy(0, 42, foe_cnt)
