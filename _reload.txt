var components
func check_weapon_name(weap, short)
  components = string.Split(short, true)
  for c : components
    ?string.Sub(c, 0, 1) = ＂-＂
      ?weap = string.Sub(c, 1)
        return false
    :
      ?weap ! c
        return false

  return true

func reload_listen()
  ?ranged.Contains(item.right) 
  ^ & item.right.state ! 4
    ranged.RemoveAt(ranged.IndexOf(item.right))
  ?melee.Contains(item.right)
  ^ & item.right.state ! 4
    melee.RemoveAt(melee.IndexOf(item.right))
  
  ?ranged.Contains(item.left)
  ^ & item.left.state ! 4
    ranged.RemoveAt(ranged.IndexOf(item.left))
  ?melee.Contains(item.left)
  ^ & item.left.state ! 4
    melee.RemoveAt(melee.IndexOf(item.left))

  ?item.right.state = 3
    ?reload_valid(item.right)
      ?reload_type(item.right) = ＂ranged＂
        ?!ranged.Contains(item.right)
          ranged.Add(item.right)
      :
        ?!melee.Contains(item.right)
          melee.Add(item.right)

  ?item.left.state = 3
    ?reload_valid(item.left)
      ?reload_type(item.left) = ＂ranged＂
        ?!ranged.Contains(item.left)
          ranged.Add(item.left)
      :
        ?!melee.Contains(item.left)
          melee.Add(item.left)     

func can_reload_melee()
  ?melee.Count() > 0
    return true
  return false

func can_reload_ranged()
  ?ranged.Count() > 0
    return true
  return false

func can_reload()
  ?can_reload_melee() | can_reload_ranged()
    return true
  return false

func need_reload(weapon)
  for m : melee
    ?check_weapon_name(m, weapon)
      return true
  for r : ranged
    ?check_weapon_name(r, weapon)
      return true
  return false

func next_weapon(weapons, num)
  ret_list.Clear()
  for weapon : weapons
    ?ret_list.Count() = num
      break

        ret_list.Clear()

        return ret_list

    ?!need_reload(weapon) | 
    ^num - (weapons.Count() -
    ^weapons.IndexOf(weapon)) >= 0
      ret_list.Add(weapon)

  return ret_list
      
      
func reload_melee()
  ?can_reload_melee()
    //reload_display(30, 1)
    ?item.right ! trisk
      equipR trisk
      equipL sword shiny
    :
      equipL trisk
      ?melee[0] ! "shiny"
        equip @melee[0] + " -shiny"@
      :
        equip @melee[0]@

func reload_ranged()
  ?can_reload_ranged()
    //reload_display(30,1)
    ?item.right ! trisk
      equipR trisk
      equipL sword shiny
    :
      equipL triskelion
      ?ranged[0] ! "shiny"
        equip @ranged[0] + " -shiny"@
      :
        equip @ranged[0]@

func reload_all()
  ?can_reload_melee()
    reload_melee()
  :?can_reload_ranged()
    reload_ranged()
