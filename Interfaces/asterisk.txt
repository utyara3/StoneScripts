func Display(x, y, color, text)
  >`@x@,@y@,#@color@,@text@


var days = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"]
func get_short_day_name(year, month, day)
  ?month < 3
    month = month + 12
    year = year - 1

  var K = year % 100
  var J = year // 100

  var h = (day + math.FloorToInt((13 * (month + 1)) / 5) + K + 
          ^math.FloorToInt(K / 4) + math.FloorToInt(J / 4) + 5 * J) % 7

  //var index = math.Floor(h)

  return days[h]

func pad(num)
  ?num < 10
    num = "0" + num

  return num

func get_time_string(hour, minute, second)
  return pad(hour) + ":" + pad(minute) + ":" + pad(second)

func display_time(x, y, color)
  var year = time.year
  var month = time.month
  var day = time.day

  var day_name = get_short_day_name(year, month, day)
  
  var day_string = year + " " + pad(month) + "/" + pad(day) +
                   ^ " " + day_name

  var time_string = get_time_string(time.hour, time.minute, time.second)

  Display(x, y, color, day_string + "  " + time_string)

func format_loc_time(frames)
  return math.Round(frames / 60)
